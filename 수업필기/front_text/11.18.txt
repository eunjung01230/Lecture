$btnRemove.addEventListener('mouseover', () => alert('호버!'));
$btnRemove.removeEventListener('mouseover', () => alert('호버!'));

const handleMouseOver = () => alert('호버!');   // 주소 A
addEventListener('mouseover', handleMouseOver);    // 주소 A 등록
removeEventListener('mouseover', handleMouseOver); // 주소 A 제거 ✅

즉, 
removeEventListener는 "등록된 함수 중에 이것과 ===인 것"을 찾아서 지움
// ❌ 매번 새 함수 → 주소가 달라서 못 찾음
btn.addEventListener('click', () => alert('hi'));      // 100번지 등록
btn.removeEventListener('click', () => alert('hi'));   // 200번지를 찾음 → 없음

// ✅ 변수에 담아 같은 함수 사용
const sayHi = () => alert('hi');                        // 100번지
btn.addEventListener('click', sayHi);                   // 100번지 등록
btn.removeEventListener('click', sayHi);                // 100번지 찾음 → 제거
