-- as : Alias (별칭) 부여하기 
-- 메뉴테이블에서 menu_name, menu_price 를 조회해줘 
-- 단, menu_name은 메뉴이름/ menu_price는 가격 
-- 이렇게 표시해
select
	menu_name as '메뉴이름',    /// 띄어쓰기나 추가가 없으면 ' ' 없어도 상관없음 
        menu_price as '가격' 		/// 단, 띄어쓰기가 있으면 오류가 뜸 
from
	tbl_menu;
-------------------------------------------------

-----------------------------------------------------
menu_name like '%마늘%'; -- %의 위칙 중요 
					중간에 들어가면 꼭 앞뒤에 같이 있어야 함 
- %단어: 단어로 끝나는 데이터를 반환 
- 단어%: 단어로 시작하는 데이터를 반환 
- %단어%: 단어를 포함하는 데이터를 반환 
--------------------------------------------------------------
-- in 연산자 
-- 메뉴테이블에서 이름, 카테고리 조회 
-- 단, 카테고리 코드가 4,5,6 인 메뉴만 조회 
select
	menu_name, 
    category_code
from 
	tbl_menu
where 
	category_code = 4
    or
    category_code = 5
    or
    category_code = 6;

== category_code in (4,5,6,); -- = category_code = 4 or category_code = 5 or category_code = 6;
