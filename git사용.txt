# 1. 처음 한 번만 clone
git clone 저장소주소

# 2. clone한 폴더로 이동
cd 프로젝트폴더

# 3. 최신 내용 받아오기
git pull origin main

# 4. 내 작업용 브랜치 생성 + 이동
git switch -c 내브랜치이름

# 예:
git switch -c feature/login

# 5. 파일 작업 후 변경 확인
git status

# 6. 올릴 파일 추가
git add .

# 또는 특정 파일만
git add 파일명

# 7. 커밋
git commit -m "로그인 기능 추가"

# 8. 내 브랜치를 조직 GitHub에 push
git push -u origin 내브랜치이름

# 예:
git push -u origin feature/login
